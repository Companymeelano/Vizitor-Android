### CalcDetailsPishfactor

CREATE FUNCTION [dbo].[CalcDetailsPishfactor]
(
	@shfacfo int
)
RETURNS int
AS
BEGIN

	declare @result int = (select count(dbo.subsailfact_pish.rdf__) from subsailfact_pish where shfacfo = @shfacfo and active = 't')
	return @result

END

                        if(@Setting97>0)
                            begin
                    set @date=(select dbo.what_date(@date,@setting97))
                            end
            end
            end
            --Step 1
            insert into subsailfact (rdf__, shfacfo, SHKA, rdf_anbar, TEDVAH, TEDJOZ, VAHPRICE, JOZPRICE, BASTEBANDI, TEDBASTEBANDI, LINESUM, PERTAFIF, TafifAghlam,RDF, PERVIS, litakhma, active, naka, ptax, tax, avarez, pavarez, tedjoz_fel, tedvah_fel, tedbastebandi_fel, linesum_fel, litakhma_fel, tax_fel, avarez_fel, Gift,TafifLine,PerPromotion,PromotionValue,TafifLineFel,ProductionSeriesID,TEDVAHMain,TEDJOZMain,TafifPos,TafifNaghd,VarietyID)

            --End Step 1
            --Step 2

            if @ProductionSeriesID is not null
            begin
                    set @invepgh = (select PriceEnd from produce.ProductionSeries where ID=@ProductionSeriesID)
                end
            else
                begin
                    set @invepgh = (select inventory_price from inventory where shka=@shka)
                end

            set @Gain=dbo.cal_gain(1,@tedad,@JOZPRICE,@invepgh,@litakhma,@pervis)
            insert into ka_act(shka, act_id, shfac, rdf__, act_dis, act_date, done_date, [UserID], tedvah, tedjoz, tedbastebandi, active, price, invepgh, invep, gain, rdf_kh,  ptax, tax, vis_rdf, pertaf, pervis, litakhma, HAct_id, ghno, sysid, pavarez, avarez, Gift,RdfAnbar,shmo,PerPromotion,TafifHajmi1,ProductionSeriesID, TamamJoz,VarietyID)
            values(@shka,20,@shfacfo,@rdf__,@ActDis,@date,@DoneDate,@UserID,@TEDVAH,@TEDJOZ,@TEDBASTEBANDI,@active,@JOZPRICE,@invepgh,@invepgh,@Gain,@RDF,@ptax,@tax,@VisRdf,@PERTAFIF,@PERVIS,@litakhma,20,@shfacfo,@sysid,@pavarez,@avarez,@Gift,@rdf_anbar,@shmo,@PerPromotion,@TafifLine,@ProductionSeriesID, FLOOR(@JOZPRICE - (@litakhma/@tedad)),@VariteyID)
            --End Step 2
            print('')
            end

### pishfactor_body

create VIEW [dbo].[pishfactor_body]
AS
SELECT  dbo.inventory.naka, dbo.inventory.vahsanj, dbo.inventory.mohvah as moh, dbo.sailfact_pish.sh_f, dbo.inventory.shka, dbo.subsailfact_pish.TEDVAH as tedvah, dbo.subsailfact_pish.TEDJOZ,                        dbo.subsailfact_pish.rdf_anbar, CONVERT(decimal(18, 0), dbo.subsailfact_pish.JOZPRICE) AS jozprice, CONVERT(decimal(18, 0), dbo.subsailfact_pish.LINESUM) AS linesum,
        CAST(dbo.subsailfact_pish.PERTAFIF AS decimal) AS PERTAFIF, CAST(dbo.subsailfact_pish.PerPromotion AS decimal) AS PerPromotion,
        CAST(dbo.subsailfact_pish.PERVIS AS decimal) AS PERVIS, CAST(dbo.subsailfact_pish.Ptax AS decimal) AS pTax, CAST(dbo.subsailfact_pish.Pavarez AS decimal) AS pAvarez, CONVERT(decimal(18, 0),
        dbo.subsailfact_pish.litakhma) AS litakhma, dbo.sailfact_pish.shfacfo, dbo.sailfact_pish.ted_rooz, dbo.sailfact_pish.date, dbo.anbars.rdf_anbar AS Expr1, dbo.anbars.name, ISNULL                             (dbo.subsailfact_pish.Tax, 0) AS Tax, ISNULL(dbo.subsailfact_pish.Avarez, 0) AS Avarez,dbo.subsailfact_pish.RDF,
	  CONVERT(decimal(18, 0),((dbo.subsailfact_pish.LINESUM)-(dbo.subsailfact_pish.litakhma)) + ((dbo.subsailfact_pish.Avarez)+(dbo.subsailfact_pish.Tax))) AS LineSum_Takhfif_Tax

FROM dbo.inventory INNER JOIN dbo.subsailfact_pish ON dbo.inventory.shka = dbo.subsailfact_pish.SHKA
                   INNER JOIN dbo.sailfact_pish ON dbo.subsailfact_pish.rdf__ = dbo.sailfact_pish.rdf__ AND dbo.subsailfact_pish.shfacfo = dbo.sailfact_pish.shfacfo
				   INNER JOIN dbo.anbars ON dbo.subsailfact_pish.rdf_anbar = dbo.anbars.rdf_anbar
WHERE (dbo.sailfact_pish.active = 't')
