## AddFromAtiranDetailsForVisitors   (955 chars)

create PROCEDURE [dbo].[AddFromAtiranDetailsForVisitors]
	@docID int,
	@bed bigint,
	@bes bigint,
	@desc nvarchar(1000),
	@TafsilID int,
	@MoeinID int,
	@shfacfo int,
	@vis_rdf int,
	@KindForosh int,
	@LevelFiveID int=null,
	@LevelSixID int=null
AS
BEGIN
    
	execute AddToFromAtiranDetails @docID,@bed,@bes,@desc,@TafsilID,@MoeinID,@shfacfo,@KindForosh,@LevelFiveID,@LevelSixID

	declare @super int =(select supervisor_rdf from visitors where vis_rdf=@vis_rdf)
	if(@super <> 0)
	begin
		declare @supervisorPer float = (select supervisor_per from visitors where vis_rdf=@vis_rdf)
		set @bes = @bes * @supervisorPer
		set @bed = @bed * @supervisorPer
		set @TafsilID =(select CUSTOMERS.TafsilID from CUSTOMERS where Ecode_Vis=@super and kind != 8)
		if(@bes > 0 or @bed > 0)
		execute AddFromAtiranDetailsForVisitors @docID,@bed,@bes,@desc,@TafsilID,@MoeinID,@shfacfo,@super,@KindForosh,@LevelFiveID,@LevelSixID
		
	end
END

## FixManCustomer   (599 chars)

CREATE  proc [dbo].[FixManCustomer] (@shmo int)
    as
    set xact_abort on
    begin transaction a

        DECLARE @Man MONEY;

        -- محاسبه مانده (بدهی - بستانکاری)
        SELECT @Man = ISNULL(SUM(act_bed), 0) - ISNULL(SUM(act_bes), 0) FROM cust_act WHERE shmo = @Shmo AND (isActive <> 0 OR isActive IS NULL);

        update CUSTOMERS set man=@man where SHMO =@shmo
        if((select value from overal_setting where id=117)=1)
        begin
        exec FixTasvie @shmo
        end

        exec Fix_Sys_Mandeh_Customer @shmo
    commit transaction a
    return

CREATE proc [dbo].[SelectPriceAndTedvahForushVisitorhaByDate]
@FromDate nvarchar(10),
@Todate nvarchar(10),
@MasirRdf bigint,
@CustGroup bigint,
@KaGroup bigint
as
select  sum(VWForushKhales.kol_price_tafif)as'price',sum((VWForushKhales.tedvah_fel*inventory.mohvah)+VWForushKhales.tedjoz_fel)as'tedvah',visitors.vis_rdf,visitors.vis_name,masir.rdf_masir from VWForushKhales,inventory,kagroup,visitors,masir,CUSTOMERS,custgroup
where visitors.vis_rdf=VWForushKhales.vis_rdf
and VWForushKhales.shka=inventory.shka
and( inventory.group_rdf=kagroup.group_rdf
or inventory.group_rdf in(select group_rdf from kagroup where ParentGroupRdf=inventory.group_rdf))

and VWForushKhales.shmo=CUSTOMERS.SHMO
and masir.rdf_masir=CUSTOMERS.RDF_masir
and CUSTOMERS.RDF_masir=@MasirRdf
and CUSTOMERS.group_rdf=custgroup.group_rdf
and custgroup.group_rdf=@CustGroup
and kagroup.group_rdf=@KaGroup

and VWForushKhales.act_date between @FromDate and @Todate
group by
visitors.vis_rdf,visitors.vis_name,masir.rdf_masir
having sum(VWForushKhales.kol_price_tafif)>0

    CREATE PROCEDURE [dbo].[UpdateMojodiInventory](@shka bigint)
    AS
    BEGIN
    declare @DoVahedi int=(select value from overal_setting where id=8)
    declare @kol decimal(18,3)
    declare @mohvah int=(select mohvah from inventory where shka=@Shka)
    declare @tedbastebandi decimal(18,3)=0
    set @kol=(select sum((tedvah*@mohvah+tedjoz)*(case when act_id in(20,22,5,19,18,48,26,85,133) then -1 else 1 end)) from ka_act where shka=@Shka and active='t')
    if(@DoVahedi=1)
    begin
        set @tedbastebandi=(select sum(isnull(tedbastebandi,0)*(case when act_id in(20,22,5,19,18,48,26,85,133) then -1 else 1 end)) from ka_act where shka=@shka and active='t' )
        if @tedbastebandi is null
            set @tedbastebandi=0
    end
    ---**********************
    if @mohvah=1
        begin
            update inventory set mojkajoz=0,mojkavah=@kol,tedbastebandi=@tedbastebandi where shka=@shka
        end
    else
        begin
            if((@kol/@mohvah)>=0)
            update inventory set mojkajoz=@kol%@mohvah,mojkavah=floor(@kol/@mohvah),tedbastebandi=@tedbastebandi  where shka=@shka
            else
            update inventory set mojkajoz=@kol%@mohvah,mojkavah=-floor(abs(@kol/@mohvah)),tedbastebandi=@tedbastebandi  where shka=@shka
        end

    END
